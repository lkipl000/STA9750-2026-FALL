Introduction

The Green Transit Alliance (GTA) aims to evaluate the environmental sustainability and operational efficiency of U.S. public transit systems through its GTA Value Index. This report integrates three data sets: EIA State Electricity Profiles, NTD 2025 Energy Consumption, and NTD 2025 Service by Agency to assess energy utilization, carbon emissions, and transit service performance.

Through systematic data integration and comparative analysis, this study identifies disparities in environmental efficiency across transit agencies and establishes an empirical framework for evaluating sustainable public transportation practices.

get_eia_sep <- function(state, abbr){
  library(tidyverse)
  library(stringr)
  library(httr2)
  library(rvest)
  library(fs)
  
  state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
  
  dir_name <- fs::path("data", "mp02")
  file_name <- fs::path(dir_name, state_formatted)
  
  fs::dir_create(dir_name, recurse=TRUE)
  
  if(!fs::file_exists(file_name)){
    BASE_URL <- "https://www.eia.gov"
    RESPONSE <- request(BASE_URL) |> 
      req_url_path("electricity", 
                   "state", 
                   state_formatted) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
      ) |>   
      req_retry(
        max_tries = 4, 
      ) |>
      req_perform(file_name)
    
    resp_check_status(RESPONSE)
  }
  
  TABLE <- read_html(file_name) |> 
    html_element("table") |> 
    html_table() |>
    mutate(Item = str_to_lower(Item))
  
  if("U.S. rank" %in% colnames(TABLE)){
    TABLE <- TABLE |> rename(Rank = `U.S. rank`)
  }
  
  CO2_MWh <- TABLE |> 
    filter(Item == "carbon dioxide (lbs/mwh)") |>
    pull(Value) |> 
    str_replace_all(",", "") |>
    as.numeric()
  
  PRIMARY <- TABLE |> 
    filter(Item == "primary energy source") |> 
    pull(Rank)
  
  RATE <- TABLE |>
    filter(Item == "average retail price (cents/kwh)") |>
    pull(Value) |>
    as.numeric()
  
  GENERATION_MWh <- TABLE |>
    filter(Item == "net generation (megawatthours)") |>
    pull(Value) |>
    str_replace_all(",", "") |>
    as.numeric()
  
  data.frame(CO2_MWh               = CO2_MWh, 
             primary_source        = PRIMARY,
             electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
             # * 1000 kWh to MWH 
             generation_MWh        = GENERATION_MWh, 
             state                 = state, 
             abbreviation          = abbr
  )
}

load_ntd_energy <- function() {
  
  library(readxl)
  library(tidyverse)
  library(fs)
  
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)
  
  NTD_ENERGY_FILE <- fs::path(
    DATA_DIR, "2025_ntd_energy.xlsx"
  )
  
  # Confirm the Excel file exists
  if (!fs::file_exists(NTD_ENERGY_FILE)) {
    stop(
      "Excel file not found. Download it from the ",
      "FTA website and save it to data/mp02/"
    )
  }
  
  # Convert missing values and hyphens to zero
  to_numeric_fill_0 <- function(x) {
    x <- as.character(x)
    x[x == "-"] <- NA_character_
    replace_na(as.numeric(x), 0)
  }
  
  # Read and clean the energy dataset
  read_excel(NTD_ENERGY_FILE) |>
    select(
      -any_of(c(
        "Reporter Type",
        "Reporting Module",
        "Gallons of Other Fuel",
        "Other Fuel Description"
      ))
    ) |>
    mutate(
      across(
        -all_of(c("Agency Name", "Mode", "TOS")),
        to_numeric_fill_0
      )
    ) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(
      across(where(is.numeric), sum),
      .groups = "drop"
    ) |>
    rowwise() |>
    mutate(
      ENERGY = sum(
        c_across(
          where(is.numeric) & !all_of("NTD ID")
        ),
        na.rm = TRUE
      )
    ) |>
    ungroup() |>
    filter(ENERGY > 0) |>
    select(-ENERGY)
}

# Standardizing the NTD fuel column names.
#
# The NTD workbook uses long, inconsistent column names
# (e.g. "Gallons of Diesel Fuel", "C Natural Gas"), and the exact
# labels change between report years. Rather than hard-coding
# each one, I am matching each name against keywords and mapping
# it to a short, stable name. The order of the case_when()
# conditions matters: "bio-diesel" must be caught before "diesel",
# and the liquefied gases before compressed natural gas.

std_fuel_name <- function(x) {
  x_low <- str_to_lower(x)
  case_when(
    str_detect(x_low, "description") ~ "drop_description",
    str_detect(x_low, "bio") ~ "biodiesel",
    str_detect(x_low, "battery") ~ "elec_battery",
    str_detect(x_low, "propulsion") ~ "elec_propulsion",
    str_detect(x_low, "petroleum") ~ "lpg",
    str_detect(x_low, "liqu") & str_detect(x_low, "nat") ~ "lng",
    str_detect(x_low, "natural gas|cng") ~ "cng",
    str_detect(x_low, "diesel") ~ "diesel",
    str_detect(x_low, "gasoline") ~ "gasoline",
    str_detect(x_low, "bunker|residual") ~ "bunker",
    str_detect(x_low, "kerosene|jet") ~ "kerosene",
    str_detect(x_low, "ethanol") ~ "ethanol",
    str_detect(x_low, "metha|metho") ~ "methanol",
    str_detect(x_low, "hydrogen") ~ "hydrogen",
    .default = x
  )
}



# Recording the conversion factor for every fuel in one table.
#
# I am storing pounds of CO2 per unit and MMBtu of energy per unit
# for each fuel. Factors are from the EIA "Carbon Dioxide Emissions
# Coefficients" table (https://www.eia.gov/environment/emissions/co2_vol_mass.php)
# and EIA heat contents (Monthly Energy Review, Appendix A), aligned
# to the NTD fuel categories as follows:
#
#   - Bunker fuel -> EIA residual fuel oil
#   - Bio-diesel  -> assumed B20 blend (80% of the diesel factor),

# Aggregating the service table to one row per agency and mode.
#
# NTD_SERVICE has one row per agency, mode, AND type of service
# (directly operated vs. purchased transportation), whereas the

Total Emissions

Fuel quantities alone cannot be compared directly because gallons and kilowatt-hours represent different units and emissions profiles. I estimate operational CO₂ by applying fuel-specific emissions factors to reported consumption and assigning electricity-related emissions using the relevant state grid intensity. The resulting agency–mode estimates support subsequent comparisons, subject to the stated assumptions.

Show Code 
# Joining the three tables and computing total CO2 per agency + mode.
#
# I am building one row per agency + mode pair:
#
#   1. NTD_ENERGY_CLEAN supplies the fuel use for each pair.
#   2. An inner join to AGENCY_INFO attaches the agency's state
#      (an inner join, since an agency without service data
#      cannot be normalized in Task 6 anyway).
#   3. A left join to EIA_SEP_REPORT attaches the state's grid

Normalizing Emissions to Transit Usage

Total emissions favor small systems and penalize large ones, even when larger agencies transport substantially more passengers. To make the comparison more informative, I normalize estimated CO₂ by unlinked passenger trips and passenger miles. Agencies are also grouped by annual ridership so that comparisons reflect differences in operating scale. These intensity measures are complementary: per-trip values reflect boardings, while per-mile values account for distance traveled.

Show Code 
# Linking emissions to passenger service at the agency + mode level.
#
# I am joining on both the agency ID and the mode, so that each
# mode's emissions are matched with the riders it actually carried.

EMISSIONS_SERVICE <- EMISSIONS |>
  inner_join(
    SERVICE_AGENCY_MODE |> select(ntd_id, mode, total_upt, passenger_miles),
    join_by(ntd_id, mode)
  )



# Setting the counterfactual: every passenger mile driven instead.
#
# I am converting passenger miles to gallons of gasoline using the
# CAFE fleet-wide average, assuming one occupant per car, and then
# to pounds of CO2 with the EIA gasoline factor.
#
# CAFE_MPG is the NHTSA MY2026 fleet-wide requirement (~49 mpg,
# 87 FR 25710). That flatters cars, since the on-road fleet is
# much less efficient, so my "emissions avoided" figures are
# deliberately conservative.

CAFE_MPG <- 49
CAR_LBS_PER_MILE <- EF[["gasoline"]] / CAFE_MPG



# Summarising to the agency level and assigning size classes.
#
# Ridership is reported per agency, so I am summing emissions,
# trips, miles, and energy across each agency's modes.
#
# I am then removing agencies with zero emissions or zero
# ridership (usually purchased-transportation reporters that
# cannot be assessed), and agencies under 250,000 annual trips,
# which are too small to compare fairly.
#
# The size classes are based on annual unlinked passenger trips:
#   Small: 250k - 2.5M, Medium: 2.5M - 25M, Large: 25M and above.

AGENCY_EFFICIENCY <- EMISSIONS_SERVICE |>
  group_by(ntd_id) |>
  summarise(
    agency = first(agency),
    state = first(state),
    uza_name = first(uza_name),
    n_modes = n(),
    emissions_lbs = sum(total_emissions_lbs),
    total_upt = sum(total_upt),
    passenger_miles = sum(passenger_miles),
    total_mmbtu = sum(total_mmbtu),
    electric_mmbtu = sum(electric_mmbtu),
    grid_imputed = any(grid_imputed),
    .groups = "drop"
  ) |>
  filter(emissions_lbs > 0, total_upt > 0, passenger_miles > 0) |>
  mutate(
    lbs_per_upt = emissions_lbs / total_upt,
    lbs_per_pmt = emissions_lbs / passenger_miles,
    electric_share = electric_mmbtu / total_mmbtu,
    car_emissions_lbs = passenger_miles * CAR_LBS_PER_MILE,
    avoided_lbs = car_emissions_lbs - emissions_lbs,
    size = case_when(
      total_upt >= 25e6 ~ "Large",
      total_upt >= 2.5e6 ~ "Medium",
      total_upt >= 2.5e5 ~ "Small",
      .default = NA_character_
    ),
    size = factor(size, levels = c("Small", "Medium", "Large"))
  ) |>
  filter(!is.na(size))



# Showing the most efficient agency in each size class.
#
# I am comparing efficiency on both bases (per trip and per
# passenger mile) alongside the median agency in each class.

AGENCY_EFFICIENCY |>
  group_by(size) |>

Award Winners

The GTA awards translate the emissions analysis into distinct measures of performance. The categories recognize low emissions intensity, large estimated emissions savings, and efficient bus operations, while also identifying cases that perform poorly against the specified driving benchmark. Each award uses an explicit denominator and reference group; no single category should be interpreted as a comprehensive sustainability ranking.

Worse Than Driving (“Worst Of”)

The Worse Than Driving designation identifies the medium- or large-sized agency with the highest estimated emissions per passenger mile relative to the modeled single-occupant car benchmark. A ratio greater than 1 indicates that the agency’s estimated operating emissions exceed those of the specified driving scenario on a per-passenger-mile basis.

Port Imperial Ferry Corporation, dba: NY Waterway (NJ) records 8.66 times the benchmark, equivalent to 3.156 lbs CO₂ per passenger mile. Methodology Notes

The analysis estimates operational CO₂ emissions, rather than full lifecycle impacts from vehicle manufacturing, infrastructure, or upstream fuel production. Electricity consumption is assigned the average emissions intensity of each agency’s reported state; 4 agencies use an imputed national generation-weighted average where state-level matches are unavailable. This approach does not capture time-of-day electricity use or cross-state power purchases.

The driving counter factual assumes one occupant per vehicle and fuel economy of 49 mpg, a relatively efficient benchmark that tends to reduce modeled emissions avoided compared with a less efficient car. Bio diesel is treated as a B20 blend, and records flagged as questionable in the service data are excluded. These assumptions introduce uncertainty into agency rankings and should be considered when interpreting small differences between results. The analysis does not establish that every transit passenger would otherwise have driven alone.
